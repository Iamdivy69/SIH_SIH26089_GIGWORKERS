    23→function mulberry32(seed: number) {
    24→  let a = seed;
    25→  return () => {
    26→    a |= 0;
    27→    a = (a + 0x6d2b79f5) | 0;
    28→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    29→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    30→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    31→  };
    32→}

   659→function buildBooking(s: BookingSeed): { booking: Booking; review?: Review; transaction?: Transaction } {
   660→  const service = CATALOG_LOOKUP(s.categoryId, s.serviceId);
   661→  const charge = service.basePrice + intBetween(-2, 2) * 25;
   662→  const price = computePrice(charge);
   663→  const customer = CUSTOMERS.find((c) => c.id === s.customerId)!;
   664→  const worker = WORKERS.find((w) => w.id === s.workerId)!;
   665→  const createdAt = new Date(s.when.getTime() - intBetween(20, 72) * HOUR);
   666→  const reference = `SG-${s.id.slice(-3).toUpperCase()}${(parseInt(s.id.slice(-1), 10) + 4).toString()}`;
   667→
   668→  const baseBooking: Booking = {
   669→    id: s.id,
   670→    reference,
   671→    customerId: s.customerId,
   672→    workerId: s.workerId,
   673→    categoryId: s.categoryId,
   674→    serviceId: s.serviceId,
   675→    title: s.title,
   676→    description: s.description,
   677→    addressId: customer.addresses[0].id,
   678→    scheduledAt: iso(s.when),
   679→    durationMin: service.durationMin,
   680→    status: s.status,
   681→    paymentStatus: "authorized",
   682→    price,
   683→    matchScore: s.matchScore,
   684→    createdAt: iso(createdAt),
   685→    checklist: CHECKLISTS[s.categoryId].map((label, i) => ({
   686→      id: `chk-${s.id}-${i}`,
   687→      label,
   688→      done: ["completed", "awaiting_confirmation"].includes(s.status),
   689→    })),
   690→    evidence: [],
   691→    timeline: [{ id: `ev-${s.id}-0`, at: iso(createdAt), label: "Request created", detail: `${customer.name} · ${s.title}`, by: customer.name }],
   692→    customerNotes: s.customerNotes,
   693→  };
   694→
   695→  const review: Review | undefined = s.rating
   696→    ? {
   697→        id: `rev-${s.id}`,
   698→        bookingId: s.id,
   699→        customerId: s.customerId,
   700→        customerName: customer.name,
   701→        workerId: s.workerId,
   702→        rating: s.rating,
   703→        comment: s.reviewComment ?? "",
   704→        tags: s.reviewTags ?? [],
   705→        createdAt: iso(new Date(s.when.getTime() + 3 * HOUR)),
   706→      }
   707→    : undefined;
   708→
   709→  const transaction: Transaction | undefined =
   710→    s.status === "completed"
   711→      ? {
   712→          id: `txn-${s.id}`,
   713→          bookingId: s.id,
   714→          bookingRef: reference,
   715→          workerId: s.workerId,
   716→          date: iso(new Date(s.when.getTime() + 4 * HOUR)),
   717→          serviceTitle: s.title,
   718→          gross: price.workerGross,
   719→          platformFee: price.platformFee,
   720→          welfareContribution: price.welfareContribution,
   721→          tds: price.workerTds,
   722→          net: price.workerNetPayout,
   723→          status: rand() > 0.25 ? "credited" : "paid_out",
   724→        }
   725→      : undefined;
   726→
   727→  /* Timeline per status */
   728→  if (["completed", "awaiting_confirmation"].includes(s.status)) {
   729→    baseBooking.paymentStatus = s.status === "completed" ? "settled" : "authorized";
   730→    baseBooking.timeline.push(
   731→      { id: `ev-${s.id}-1`, at: iso(new Date(createdAt.getTime() + 20 * 60000)), label: "Worker accepted", detail: `${worker.name} accepted the job`, by: worker.name },
   732→      { id: `ev-${s.id}-2`, at: iso(new Date(createdAt.getTime() + 25 * 60000)), label: "Payment authorised", detail: "Held securely until service completion", by: "Platform" },
   733→      { id: `ev-${s.id}-3`, at: iso(new Date(s.when.getTime() - 45 * 60000)), label: "On the way", by: worker.name },
   734→      { id: `ev-${s.id}-4`, at: iso(s.when), label: "Arrived & started", by: worker.name },
   735→      { id: `ev-${s.id}-5`, at: iso(new Date(s.when.getTime() + service.durationMin * 60000)), label: "Service completed", detail: "Checklist and evidence recorded", by: worker.name },
   736→    );
   737→    if (s.status === "completed") {
   738→      baseBooking.timeline.push({ id: `ev-${s.id}-6`, at: iso(new Date(s.when.getTime() + (service.durationMin + 30) * 60000)), label: "Payment settled to worker", detail: "Breakdown available in worker earnings", by: "Platform" });
   739→      if (review) baseBooking.timeline.push({ id: `ev-${s.id}-7`, at: review.createdAt, label: "Rated by customer", detail: `${review.rating} / 5`, by: customer.name });
   740→    }
   741→    baseBooking.evidence = [
   742→      { phase: "before", capturedAt: iso(s.when), label: "Fault photographed before repair" },
   743→      { phase: "after", capturedAt: iso(new Date(s.when.getTime() + service.durationMin * 60000)), label: "Completed work after repair" },
   744→    ];
   745→  } else if (s.status === "cancelled") {
   746→    baseBooking.paymentStatus = "refunded";
   747→    baseBooking.cancellationReason = "Customer travelling — cancelled 2 days before the slot (free cancellation window)";
   748→    baseBooking.timeline.push(
   749→      { id: `ev-${s.id}-1`, at: iso(new Date(createdAt.getTime() + 2 * HOUR)), label: "Cancelled by customer", detail: "Full refund issued to source account", by: customer.name },
   750→    );
   751→  } else if (s.status === "confirmed") {
   752→    baseBooking.timeline.push(
   753→      { id: `ev-${s.id}-1`, at: iso(new Date(createdAt.getTime() + 18 * 60000)), label: "Worker accepted", by: worker.name },
   754→      { id: `ev-${s.id}-2`, at: iso(new Date(createdAt.getTime() + 22 * 60000)), label: "Payment authorised", detail: "Held securely until service completion", by: "Platform" },
   755→    );
   756→  } else if (s.status === "pending_acceptance") {
   757→    /* offer only */
   758→  } else if (s.status === "in_progress") {
   759→    baseBooking.timeline.push(
   760→      { id: `ev-${s.id}-1`, at: iso(new Date(createdAt.getTime() + 15 * 60000)), label: "Worker accepted", by: worker.name },
   761→      { id: `ev-${s.id}-2`, at: iso(new Date(createdAt.getTime() + 19 * 60000)), label: "Payment authorised", by: "Platform" },
   762→      { id: `ev-${s.id}-3`, at: iso(new Date(s.when.getTime() - 40 * 60000)), label: "On the way", by: worker.name },
   763→      { id: `ev-${s.id}-4`, at: iso(s.when), label: "Arrived & started", by: worker.name },
   764→    );
   765→    baseBooking.checklist = baseBooking.checklist.map((c, i) => ({ ...c, done: i < 2 }));
   766→    baseBooking.evidence = [{ phase: "before", capturedAt: iso(s.when), label: "Kitchen — before cleaning" }];
   767→  }
   768→
   769→  return { booking: baseBooking, review, transaction };
   770→}
   771→
   772→function CATALOG_LOOKUP(categoryId: Booking["categoryId"], serviceId: string) {
   773→  const svc = serviceById(categoryId, serviceId);
   774→  if (!svc) throw new Error(`Unknown service ${serviceId} in ${categoryId}`);
   775→  return svc;
   776→}
