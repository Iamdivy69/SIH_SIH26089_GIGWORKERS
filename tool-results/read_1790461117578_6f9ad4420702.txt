    43→type Handler = (req: NextRequest, params: Params, body: any, user: string) => Promise<unknown> | unknown;
    44→
    45→const routes: { method: string; pattern: RegExp; keys: string[]; handler: Handler }[] = [];
    46→
    47→function route(method: string, path: string, handler: Handler) {
    48→  const keys: string[] = [];
    49→  const pattern = new RegExp(
    50→    "^" +
    51→      path
    52→        .split("/")
    53→        .map((seg) => {
    54→          if (seg.startsWith(":")) {
    55→            keys.push(seg.slice(1));
    56→            return "([^/]+)";
    57→          }
    58→          return seg.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    59→        })
    60→        .join("/") +
    61→      "$",
    62→  );
    63→  routes.push({ method, pattern, keys, handler });
    64→}
    65→
    66→export async function handle(req: NextRequest, slug: string[]): Promise<Response> {
    67→  await new Promise((r) => setTimeout(r, LATENCY));
    68→  const path = "/" + slug.join("/");
    69→  const method = req.method.toUpperCase();
    70→  const user = req.headers.get("x-demo-user") || "";
    71→
    72→  let body: any = null;
    73→  if (method === "POST" || method === "PUT") {
    74→    try {
    75→      body = await req.json();
    76→    } catch {
    77→      body = {};
    78→    }
    79→  }
    80→
    81→  for (const r of routes) {
    82→    if (r.method !== method) continue;
    83→    const match = path.match(r.pattern);
    84→    if (!match) continue;
    85→    const params: Params = {};
    86→    r.keys.forEach((k, i) => (params[k] = decodeURIComponent(match[i + 1])));
    87→    try {
    88→      const result = await r.handler(req, params, body, user || demoUserFor(path));
    89→      return Response.json(result as any, { status: 200 });
    90→    } catch (err: any) {
    91→      return Response.json({ error: err?.message ?? "Request failed" }, { status: 400 });
    92→    }
    93→  }
    94→  return Response.json({ error: "Not found", path }, { status: 404 });
    95→}

   185→    categoryId: customer.preferredCategories[0],

   187→    customerLocality: customer.locality,

   207→route("GET", "/workers", (req) => {
   208→  const store = getStore();
   209→  const url = new URL(req.url);
   210→  const q = (url.searchParams.get("q") ?? "").toLowerCase();
   211→  const category = url.searchParams.get("category");
   212→  const minRating = parseFloat(url.searchParams.get("minRating") ?? "0");
   213→  const maxDistance = parseFloat(url.searchParams.get("maxDistance") ?? "999");
   214→  const verifiedOnly = url.searchParams.get("verifiedOnly") === "true";
   215→  const availableToday = url.searchParams.get("availableToday") === "true";
   216→  const sort = url.searchParams.get("sort") ?? "recommended";
   217→
   218→  let list = store.workers.filter((w) => w.status === "verified" || w.status === "suspended");
   219→  if (q) list = list.filter((w) => [w.name, w.tradeTitle, w.locality, ...w.skills].join(" ").toLowerCase().includes(q));
   220→  if (category && category !== "all") list = list.filter((w) => w.category === category);
   221→  if (minRating) list = list.filter((w) => w.rating >= minRating);
   222→  if (maxDistance < 999) list = list.filter((w) => w.distanceKm <= maxDistance);
   223→  if (verifiedOnly) list = list.filter((w) => w.status === "verified");
   224→  const today = new Date().getDay();
   225→  if (availableToday) list = list.filter((w) => w.availability.some((a) => a.day === today && a.slots.length > 0));
   226→
   227→  if (sort === "rating") list.sort((a, b) => b.rating - a.rating);
   228→  else if (sort === "distance") list.sort((a, b) => a.distanceKm - b.distanceKm);
   229→  else if (sort === "jobs") list.sort((a, b) => b.completedJobs - a.completedJobs);
   230→  else list.sort((a, b) => b.rating * 20 + b.completedJobs / 20 - (a.rating * 20 + a.completedJobs / 20));
   231→
   232→  return { items: list, total: list.length };
   233→});

   255→route("POST", "/matching", (_req, _p, body) => {
   256→  const store = getStore();
   257→  /* Input hardening — same contract as POST /bookings: clear typed errors. */
   258→  const categoryId = body?.categoryId as string | undefined;
   259→  if (!categoryId || !categories().some((c) => c.id === categoryId)) {
   260→    throw new Error("Unknown service category — pick one of the six cooperative categories");
   261→  }
   262→  if (body?.serviceId !== undefined && !serviceById(categoryId as ServiceCategoryId, body.serviceId)) {
   263→    throw new Error(`Unknown service — ${String(body.serviceId)} is not offered under ${categoryId}`);
   264→  }
   265→  const charge = body?.charge !== undefined ? Number(body.charge) : 800;
   266→  if (!Number.isFinite(charge) || charge <= 0) {
   267→    throw new Error("Invalid service charge — a positive rupee amount is required");
   268→  }
   269→  const customer = customerById(CUSTOMER_USER);
   270→  const scheduledAt = new Date(body.scheduledAt ?? Date.now() + 86400000);
   271→  if (Number.isNaN(scheduledAt.getTime())) throw new Error("Invalid slot — please pick a date and time");
   272→  const recs = recommend(store.workers, {
   273→    categoryId: categoryId as ServiceCategoryId,
   274→    serviceId: body.serviceId,
   275→    scheduledAt,
   276→    customerLocality: customer.locality,
   277→    serviceCharge: charge,
   278→  });
   279→  return {
   280→    recommendations: recs,
   281→    weights: { skill: 35, location: 20, availability: 20, rating: 15, experience: 10 },
   282→    estimate: computePrice(charge),
   283→  };

   291→    throw new Error("Unknown service category — pick one of the six cooperative categories");

   296→  }

   298→  if (!worker) {
   299→    throw new Error("Unknown member — this worker does not exist");
   300→  }
   301→  const charge = Number(body?.charge);
   302→  if (!Number.isFinite(charge) || charge <= 0) {
   303→    throw new Error("Invalid service charge — a positive rupee amount is required");
   304→  }
   305→  const description = String(body?.description ?? "").trim();
   306→  if (description.length < 15) {
   307→    throw new Error("Add a short description of the work (at least 15 characters)");

   310→  const addressId = String(body?.addressId ?? customer.addresses[0].id);
   311→  if (!customer.addresses.some((a) => a.id === addressId)) {
   312→    throw new Error("Unknown service address — pick one of the customer's saved addresses");
   313→  }
   314→  const recurrence = body?.recurrence as "weekly" | "monthly" | undefined;
   315→  if (recurrence !== undefined && recurrence !== "weekly" && recurrence !== "monthly") {
   316→    throw new Error("Invalid recurrence — use \"weekly\" or \"monthly\"");
   317→  }
   318→  const scheduledAt = new Date(body.scheduledAt);
   319→  if (Number.isNaN(scheduledAt.getTime())) throw new Error("Invalid slot — please pick a date and time");

   321→    customerId: CUSTOMER_USER,
   322→    workerId: worker.id,
   323→    categoryId: categoryId as Booking["categoryId"],
   324→    serviceId: service.id,
   325→    description,
   326→    addressId,
   327→    scheduledAt: scheduledAt.toISOString(),
   328→    customerNotes: body.customerNotes,
   329→    matchScore: body.matchScore,
   330→    charge,
   331→    recurrence,
   332→  });
   333→  return { booking };
   334→});
