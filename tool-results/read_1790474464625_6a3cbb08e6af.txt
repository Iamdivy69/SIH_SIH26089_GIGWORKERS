     1→"use client";
     2→
     3→/**
     4→ * Surplus & dividends — the cooperative's annual surplus allocation builder.
     5→ *
     6→ * The showcase of cooperative economics: the surplus earned from platform
     7→ * processing fees is drafted by the admin desk, then approved by member vote;
     8→ * dividends follow patronage (completed service value), never share count.
     9→ *
    10→ * Draft state: steppers with policy floors, live stacked preview bar, live
    11→ * dividend figures, save/reset, and "Send to member vote". In-vote state:
    12→ * allocation locked, proposal banner with a deep link to governance.
    13→ */
    14→
    15→import { useMemo, useState } from "react";
    16→import { CheckCircle2, Coins, Minus, Plus, RotateCcw, Save, Vote } from "lucide-react";
    17→import { toast } from "sonner";
    18→import { Button } from "@/components/ui/button";
    19→import { Skeleton } from "@/components/ui/skeleton";
    20→import {
    21→  AlertBanner,
    22→  DataTable,
    23→  ErrorState,
    24→  PageHeader,
    25→  PersonAvatar,
    26→  SectionCard,
    27→} from "@/components/shared";
    28→import type { Column } from "@/components/shared";
    29→import { FineNote, KpiStrip, KpiStripSkeleton } from "./ui";
    30→import { useAdminSurplus, useSubmitSurplusProposal, useUpdateSurplusAllocation } from "@/hooks/use-api";
    31→import { dateFull, dateShort, money, moneyCompact, num } from "@/lib/format";
    32→import type { DividendDistribution, SurplusAllocationKey, SurplusAllocationLine, SurplusMemberPreview } from "@/lib/types";
    33→import { useAppStore } from "@/store/app-store";
    34→import { cn } from "@/lib/utils";
    35→
    36→type Pcts = Record<SurplusAllocationKey, number>;
    37→
    38→/** The board's default split — also the seeded allocation (35/40/15/7/3). */
    39→const BOARD_DEFAULT: Pcts = { reserves: 35, dividend: 40, training: 15, community: 7, contingency: 3 };
    40→
    41→/** Tone → semantic background token (allocation dot + stacked-bar segment). */
    42→const TONE_BG: Record<SurplusAllocationLine["tone"], string> = {
    43→  primary: "bg-primary",
    44→  success: "bg-success",
    45→  warning: "bg-warning",
    46→  info: "bg-info",
    47→  neutral: "bg-muted-foreground/40",
    48→};
    49→
    50→/** Column template for the 6-cell KPI strip — fills cleanly at every breakpoint. */
    51→const KPI_COLS = "sm:grid-cols-3 xl:grid-cols-6 2xl:grid-cols-6";
    52→
    53→export function AdminSurplusScreen() {
    54→  const navigate = useAppStore((s) => s.navigate);
    55→  const surplus = useAdminSurplus();
    56→  const save = useUpdateSurplusAllocation();
    57→  const submit = useSubmitSurplusProposal();
    58→
    59→  /* Local draft — null until the first edit, otherwise mirrors the server. */
    60→  const [override, setOverride] = useState<Pcts | null>(null);
    61→  const serverPcts = useMemo<Pcts | null>(
    62→    () =>
    63→      surplus.data
    64→        ? (Object.fromEntries(surplus.data.allocations.map((a) => [a.key, a.pct])) as Pcts)
    65→        : null,
    66→    [surplus.data],
    67→  );
    68→  const pcts = override ?? serverPcts;
    69→
    70→  if (surplus.isError) {
    71→    return (
    72→      <>
    73→        <PageHeader eyebrow="Cooperative" title="Surplus & dividends" />
    74→        <ErrorState message="The surplus plan could not be loaded." onRetry={() => surplus.refetch()} />
    75→      </>
    76→    );
    77→  }
    78→
    79→  if (surplus.isLoading || !surplus.data || !pcts) {
    80→    return (
    81→      <>
    82→        <PageHeader
    83→          eyebrow="Cooperative"
    84→          title="Surplus & dividends"
    85→          description="FY 2026-27 allocation of the cooperative's surplus."
    86→        />
    87→        <KpiStripSkeleton count={6} className={cn("mb-6", KPI_COLS)} />
    88→        <div className="space-y-6">
    89→          <Skeleton className="h-48 w-full rounded-lg" />
    90→          <Skeleton className="h-[26rem] w-full rounded-lg" />
    91→          <Skeleton className="h-36 w-full rounded-lg" />
    92→        </div>
    93→      </>
    94→    );
    95→  }
    96→
    97→  const d = surplus.data;
    98→  const inVote = d.status === "in_vote";
    99→  const distributed = d.status === "distributed";
   100→  const locked = inVote || distributed;
   101→  const sum = d.allocations.reduce((acc, l) => acc + pcts[l.key], 0);
   102→  const balanced = sum === 100;
   103→  const dirty = d.allocations.some((l) => pcts[l.key] !== l.pct);
   104→  const livePool = Math.round((d.surplusYtd * pcts.dividend) / 100);
   105→  const liveAvg = d.sharingMembers > 0 ? Math.round(livePool / d.sharingMembers) : 0;
   106→  const amountFor = (pct: number) => money(Math.round((d.surplusYtd * pct) / 100));
   107→
   108→  const setPct = (line: SurplusAllocationLine, next: number) => {
   109→    setOverride({ ...pcts, [line.key]: Math.max(line.minPct, Math.min(100, next)) });
   110→  };
   111→
   112→  /* Dividend preview follows the live draft split — same math as the server. */
   113→  const previewRows: SurplusMemberPreview[] = d.memberPreview.slice(0, 10).map((m) => ({
   114→    ...m,
   115→    dividend: Math.round((livePool * m.patronage) / Math.max(1, d.patronageTotal)),
   116→  }));
   117→
   118→  const onSave = () => {
   119→    if (!balanced) return;
   120→    save.mutate(pcts, { onError: (e) => toast.error(e.message) });
   121→  };
   122→
   123→  const onSend = () => {
   124→    if (!balanced || dirty) return;
   125→    submit.mutate(undefined, { onError: (e) => toast.error(e.message) });
   126→  };
   127→
   128→  const ld = d.lastDistributed;
   129→
   130→  return (
   131→    <>
   132→      <PageHeader
   133→        eyebrow="Cooperative"
   134→        title="Surplus & dividends"
   135→        description={
   136→          distributed
   137→            ? `FY ${d.fiscalYear} allocation · distributed ${dateShort(d.lastDistributed.distributedAt)} with member approval`
   138→            : `FY ${d.fiscalYear} allocation · ${money(d.surplusYtd)} surplus to distribute with member approval`
   139→        }
   140→      />
   141→
   142→      <KpiStrip
   143→        className={cn("mb-6", KPI_COLS)}
   144→        cells={[
   145→          {
   146→            label: "Surplus YTD",
   147→            value: money(d.surplusYtd),
   148→            sub: `platform fees ${money(d.platformFeesYtd)} − costs ${money(d.operatingCostsYtd)}`,
   149→          },
   150→          {
   151→            label: "Dividend pool",
   152→            value: money(livePool),
   153→            sub: distributed ? "distributed to members" : "at current allocation",
   154→            tone: "positive",
   155→          },
   156→          { label: "Avg dividend", value: money(liveAvg), sub: distributed ? "credited per sharing member" : "per sharing member" },
   157→          { label: "Patronage base", value: money(d.patronageTotal), sub: "completed service value FY" },
   158→          { label: "Sharing members", value: num(d.sharingMembers), sub: "members with completed work" },
   159→          { label: "Platform fees YTD", value: money(d.platformFeesYtd), sub: "6% processing share, settled" },
   160→        ]}
   161→      />
   162→
   163→      <div className="space-y-6">
   164→        {/* ---------------- how this year's surplus is built ---------------- */}
   165→        <SectionCard
   166→          title="How this year's surplus is built"
   167→          description={`FY ${d.fiscalYear} to date — every rupee traceable to settled bookings.`}
   168→        >
   169→          <dl>
   170→            <div className="flex items-baseline justify-between gap-4 border-b border-border/60 py-2.5">
   171→              <div className="min-w-0">
   172→                <dt className="text-sm">Platform processing fees</dt>
   173→                <p className="mt-0.5 text-xs text-muted-foreground">6% cooperative share of settled bookings</p>
   174→              </div>
   175→              <dd className="tnum shrink-0 text-sm font-medium">{money(d.platformFeesYtd)}</dd>
   176→            </div>
   177→            <div className="flex items-baseline justify-between gap-4 py-2.5">
   178→              <div className="min-w-0">
   179→                <dt className="text-sm">Less operating costs</dt>
   180→                <p className="mt-0.5 text-xs text-muted-foreground">
   181→                  centre, staff, tool bank, training stipends — simulated
   182→                </p>
   183→              </div>
   184→              <dd className="tnum shrink-0 text-sm font-medium">
   185→                <span className="mr-1 text-muted-foreground">−</span>
   186→                {money(d.operatingCostsYtd)}
   187→              </dd>
   188→            </div>
   189→            <div className="flex items-baseline justify-between gap-4 border-t pt-3">
   190→              <dt className="text-sm font-semibold">Surplus for the year</dt>
   191→              <dd className="tnum shrink-0 text-base font-semibold tracking-tight">
   192→                <span className="mr-1 font-normal text-muted-foreground">=</span>
   193→                {money(d.surplusYtd)}
   194→              </dd>
   195→            </div>
   196→          </dl>
   197→          <FineNote className="mt-4 border-t pt-3">
   198→            Operating costs are simulated at 62% of platform-fee income in this prototype. The fee figure above reconciles
   199→            line-for-line with the commission ledger on the Finance screen — the same settled transactions, viewed from the
   200→            cooperative side. The surplus accrues as bookings settle.
   201→          </FineNote>
   202→        </SectionCard>
   203→
   204→        {/* ---------------- the allocation builder ---------------- */}
   205→        <SectionCard
   206→          title="Proposed allocation"
   207→          description={
   208→            distributed
   209→              ? "This split was approved by the members and executed — a read-only record of the FY distribution."
   210→              : inVote
   211→                ? "This is the split the members are voting on — editing is locked while the vote is open."
   212→                : "Adjust the split — the bar and dividend figures update live. Floors are member-approved guardrails, and the total must balance at 100% before saving."
   213→          }
   214→          actions={
   215→            !locked && (
   216→              <Button
   217→                size="sm"
   218→                onClick={onSend}
   219→                disabled={!balanced || dirty || submit.isPending}
   220→                title={
   221→                  !balanced
   222→                    ? "Allocation must total 100% before it can be voted on"
   223→                    : dirty
   224→                      ? "Save the draft first — members vote on the saved split"
   225→                      : "Publish this allocation as a proposal for all 216 members"
   226→                }
   227→              >
   228→                <Vote className="mr-1.5 h-3.5 w-3.5" strokeWidth={1.9} />
   229→                {submit.isPending ? "Sending…" : "Send to member vote"}
   230→              </Button>
   231→            )
   232→          }
   233→        >
   234→          {distributed && (
   235→            <div
   236→              className="mb-5 flex flex-col gap-3 rounded-lg border border-success/40 bg-success-muted/60 px-4 py-3.5 sm:flex-row sm:items-center sm:justify-between"
   237→              role="status"
   238→            >
   239→              <p className="flex min-w-0 items-start gap-2.5 text-[13px] leading-relaxed">
   240→                <CheckCircle2 className="mt-0.5 h-4 w-4 shrink-0 text-success-deep" strokeWidth={1.9} aria-hidden />
   241→                <span>
   242→                  <span className="font-semibold text-success-deep">
   243→                    FY {d.fiscalYear} distribution executed {dateFull(d.lastDistributed.distributedAt)}.
   244→                  </span>{" "}
   245→                  {money(d.dividendPool)} patronage dividend across {num(d.lastDistributed.members)} members — every rupee credited
   246→                  patronage-proportionally and recorded in the ledger below and the audit log.
   247→                </span>
   248→              </p>
   249→              <Button variant="outline" size="sm" className="shrink-0" onClick={() => navigate("admin-audit")}>
   250→                Audit trail
   251→              </Button>
   252→            </div>
   253→          )}
   254→
   255→          {inVote && d.proposal && (
   256→            <AlertBanner
   257→              severity="warning"
   258→              className="mb-5"
   259→              title={`${d.proposal.code} is with the members — voting closes ${dateFull(d.proposal.closesAt)}`}
   260→              detail="The allocation takes effect only if the proposal passes."
   261→              action="View proposal"
   262→              onAction={() => navigate("admin-governance")}
   263→            />
   264→          )}
   265→
   266→          <ul className="divide-y">
   267→            {d.allocations.map((line) => (
   268→              <AllocationRow
   269→                key={line.key}
   270→                line={line}
   271→                pct={pcts[line.key]}
   272→                amount={amountFor(pcts[line.key])}
   273→                locked={locked}
   274→                onStep={(next) => setPct(line, next)}
   275→              />
   276→            ))}
   277→          </ul>
   278→
   279→          {/* live stacked allocation preview — updates before saving */}
   280→          <div className="mt-6">
   281→            <div
   282→              className="flex h-3 w-full overflow-hidden rounded-full bg-muted"
   283→              role="img"
   284→              aria-label={`Allocation preview: ${d.allocations.map((l) => `${l.label} ${pcts[l.key]}%`).join(", ")}`}
   285→            >
   286→              {d.allocations.map((l) => (
   287→                <div key={l.key} className={cn("h-full", TONE_BG[l.tone])} style={{ width: `${pcts[l.key]}%` }} />
   288→              ))}
   289→            </div>
   290→            <ul className="mt-3 flex flex-wrap gap-x-6 gap-y-2">
   291→              {d.allocations.map((l) => (
   292→                <li key={l.key} className="flex min-w-0 items-center gap-2 text-xs">
   293→                  <span className={cn("h-2 w-2 shrink-0 rounded-full", TONE_BG[l.tone])} aria-hidden />
   294→                  <span className="truncate text-muted-foreground">{l.label}</span>
   295→                  <span className="tnum shrink-0 font-medium">{pcts[l.key]}%</span>
   296→                  <span className="tnum shrink-0 text-muted-foreground">{amountFor(pcts[l.key])}</span>
   297→                </li>
   298→              ))}
   299→            </ul>
   300→          </div>
   301→
   302→          {inVote ? (
   303→            <div className="mt-6 space-y-2 border-t pt-4">
   304→              <div className="flex flex-wrap items-center gap-2.5">
   305→                <Button variant="outline" size="sm" disabled title="Locked while the member vote is open">
   306→                  <Vote className="mr-1.5 h-3.5 w-3.5" strokeWidth={1.9} />
   307→                  Send to member vote
   308→                </Button>
   309→                <p className="text-xs text-muted-foreground">Already sent — members are voting on this allocation now.</p>
   310→              </div>
   311→              <FineNote>
   312→                A new draft can be opened after the vote closes{d.proposal ? ` (${dateFull(d.proposal.closesAt)})` : ""}.
   313→              </FineNote>
   314→            </div>
   315→          ) : distributed ? (
   316→            <div className="mt-6 space-y-2 border-t pt-4">
   317→              <div className="flex flex-wrap items-center gap-2.5">
   318→                <Button variant="outline" size="sm" disabled title="This FY's allocation has been executed">
   319→                  <Vote className="mr-1.5 h-3.5 w-3.5" strokeWidth={1.9} />
   320→                  Send to member vote
   321→                </Button>
   322→                <p className="text-xs text-muted-foreground">FY {d.fiscalYear} is complete — the split above is the executed record.</p>
   323→              </div>
   324→              <FineNote>
   325→                The next cycle opens a fresh draft after FY close. Editing is disabled on an executed allocation — the record is
   326→                certified in the audit log.
   327→              </FineNote>
   328→            </div>
   329→          ) : (
   330→            <div className="mt-6 flex flex-col gap-3 border-t pt-4 sm:flex-row sm:items-center sm:justify-between">
   331→              <div className="flex flex-wrap items-center gap-3">
   332→                <span
   333→                  className={cn(
   334→                    "micro-label whitespace-nowrap rounded-sm border px-2 py-1",
   335→                    balanced
   336→                      ? "border-success/40 bg-success-muted text-success-deep"
   337→                      : "border-destructive/40 bg-destructive-muted text-destructive-deep",
   338→                  )}
   339→                >
   340→                  {balanced
   341→                    ? "100% · balanced"
   342→                    : `${sum}% · ${sum < 100 ? `${100 - sum}% unallocated` : `${sum - 100}% over-allocated`}`}
   343→                </span>
   344→                {dirty && balanced && (
   345→                  <span className="text-xs text-muted-foreground">
   346→                    Unsaved changes — save the draft before sending it to the members.
   347→                  </span>
   348→                )}
   349→              </div>
   350→              <div className="flex flex-wrap items-center gap-2">
   351→                <Button
   352→                  variant="outline"
   353→                  size="sm"
   354→                  onClick={() => setOverride({ ...BOARD_DEFAULT })}
   355→                  disabled={save.isPending}
   356→                >
   357→                  <RotateCcw className="mr-1.5 h-3.5 w-3.5" strokeWidth={1.9} />
   358→                  Reset to board default
   359→                </Button>
   360→                <Button size="sm" onClick={onSave} disabled={!balanced || save.isPending}>
   361→                  <Save className="mr-1.5 h-3.5 w-3.5" strokeWidth={1.9} />
   362→                  {save.isPending ? "Saving…" : "Save draft"}
   363→                </Button>
   364→              </div>
   365→            </div>
   366→          )}
   367→        </SectionCard>
   368→
   369→        {/* ---------------- member dividend preview / distribution ledger ---------------- */}
   370→        {distributed ? (
   371→          <SectionCard
   372→            title="Distribution ledger — executed"
   373→            description={`FY ${d.fiscalYear} dividends as credited, patronage-proportional, ${dateFull(d.lastDistributed.distributedAt)}. Immutable record — mirrors each member's own dividend history.`}
   374→            forTable
   375→          >
   376→            <DataTable
   377→              columns={ledgerColumns}
   378→              rows={d.distributionLedger}
   379→              getRowKey={(r) => r.id}
   380→              emptyTitle="No distribution records"
   381→              emptyDescription="Records appear here the moment a vote passes and the dividend executes."
   382→              mobileCard={(r) => (
   383→                <div className="space-y-2">
   384→                  <div className="flex items-center gap-2.5">
   385→                    <PersonAvatar name={r.name} size="sm" />
   386→                    <div className="min-w-0 flex-1">
   387→                      <p className="truncate text-[13px] font-medium">{r.name}</p>
   388→                      <p className="tnum truncate font-mono text-[11px] text-muted-foreground">{r.reference}</p>
   389→                    </div>
   390→                    <p className="tnum shrink-0 text-sm font-semibold text-success-deep">{money(r.amount)}</p>
   391→                  </div>
   392→                  <p className="tnum flex items-center justify-between pl-10 text-xs text-muted-foreground">
   393→                    <span>Patronage {money(r.patronage)}</span>
   394→                    <span>Share {r.sharePct.toFixed(1)}%</span>
   395→                  </p>
   396→                </div>
   397→              )}
   398→            />
   399→            <div className="border-t px-5 py-3">
   400→              <FineNote>
   401→                {num(d.distributionLedger.length)} records · pool {money(d.dividendPool)} · avg {money(liveAvg)} per member. Each
   402→                member was notified on execution and sees this entry in their Earnings → patronage-dividend history.
   403→              </FineNote>
   404→            </div>
   405→          </SectionCard>
   406→        ) : (
   407→          <SectionCard
   408→            title="Member dividend preview"
   409→            description="Dividends follow patronage — the completed service value each member delivered this year — never share count. One member, one vote."
   410→            forTable
   411→          >
   412→          <DataTable
   413→            columns={memberColumns}
   414→            rows={previewRows}
   415→            getRowKey={(m) => m.workerId}
   416→            emptyTitle="No members with completed work yet"
   417→            emptyDescription="Members appear here as bookings settle through the year."
   418→            mobileCard={(m) => (
   419→              <div className="space-y-2">
   420→                <div className="flex items-center gap-2.5">
   421→                  <PersonAvatar name={m.name} size="sm" />
   422→                  <div className="min-w-0 flex-1">
   423→                    <p className="truncate text-[13px] font-medium">{m.name}</p>
   424→                    <p className="truncate text-xs text-muted-foreground">{m.trade}</p>
   425→                  </div>
   426→                  <p className="tnum shrink-0 text-sm font-semibold text-success-deep">{money(m.dividend)}</p>
   427→                </div>
   428→                <p className="tnum flex items-center justify-between pl-10 text-xs text-muted-foreground">
   429→                  <span>Patronage {money(m.patronage)}</span>
   430→                  <span>Share {m.sharePct.toFixed(1)}%</span>
   431→                </p>
   432→              </div>
   433→            )}
   434→          />
   435→            <div className="border-t px-5 py-3">
   436→              <FineNote>
   437→                Preview at the current dividend percentage — final figures are set by the member vote. Top{" "}
   438→                {num(previewRows.length)} of {num(d.sharingMembers)} sharing members by patronage; every member with
   439→                completed work this year shares in the distribution.
   440→              </FineNote>
   441→            </div>
   442→          </SectionCard>
   443→        )}
   444→
   445→        {/* ---------------- last year's distribution ---------------- */}
   446→        <SectionCard
   447→          title={`Most recent completed distribution — FY ${ld.fiscalYear}`}
   448→          description="The most recent completed distribution to member-owners."
   449→        >
   450→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   451→            <div>
   452→              <p className="micro-label">Surplus generated</p>
   453→              <p className="tnum mt-1 text-xl font-semibold tracking-tight">{moneyCompact(ld.surplus)}</p>
   454→              <p className="tnum mt-0.5 text-xs text-muted-foreground">{money(ld.surplus)}</p>
   455→            </div>
   456→            <div>
   457→              <p className="micro-label">Patronage bonus distributed</p>
   458→              <p className="tnum mt-1 text-xl font-semibold tracking-tight text-success-deep">
   459→                {moneyCompact(ld.patronageBonus)}
   460→              </p>
   461→              <p className="tnum mt-0.5 text-xs text-muted-foreground">
   462→                {money(ld.patronageBonus)} · 50% of surplus
   463→              </p>
   464→            </div>
   465→            <div>
   466→              <p className="micro-label">Members sharing</p>
   467→              <p className="tnum mt-1 text-xl font-semibold tracking-tight">{num(ld.members)}</p>
   468→              <p className="mt-0.5 text-xs text-muted-foreground">distributed {dateFull(ld.distributedAt)}</p>
   469→            </div>
   470→          </div>
   471→          <FineNote className="mt-4 border-t pt-3">
   472→            Dividends were paid in proportion to each member&apos;s completed service value — patronage, not share count —
   473→            after the split was approved by member vote.
   474→          </FineNote>
   475→        </SectionCard>
   476→
   477→        {/* ---------------- cooperative framing ---------------- */}
   478→        <div className="flex items-start gap-2.5 rounded-lg border bg-muted/30 p-4">
   479→          <Coins className="mt-0.5 h-4 w-4 shrink-0 text-muted-foreground" strokeWidth={1.9} />
   480→          <FineNote>
   481→            This surplus belongs to the cooperative&apos;s member-owners — the admin desk drafts the allocation, and the
   482→            members decide by vote before anything is distributed. Floors and quorum rules are set in the bye-laws, and
   483→            every change is recorded in the audit log.
   484→          </FineNote>
