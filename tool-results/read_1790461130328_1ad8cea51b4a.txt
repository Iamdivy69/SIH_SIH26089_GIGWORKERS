    24→function mulberry32(seed: number) {
    25→  let a = seed;
    26→  return () => {
    27→    a |= 0;
    28→    a = (a + 0x6d2b79f5) | 0;
    29→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    30→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    31→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    32→  };
    33→}

   758→  const worker = WORKERS.find((w) => w.id === s.workerId)!;
   759→  const createdAt =
   760→    s.createdHoursBefore !== undefined
   761→      ? new Date(s.when.getTime() - s.createdHoursBefore * HOUR)
   762→      : new Date(s.when.getTime() - intBetween(20, 72) * HOUR);

   764→
   765→  const baseBooking: Booking = {
   766→    id: s.id,
   767→    reference,
   768→    customerId: s.customerId,
   769→    workerId: s.workerId,
   770→    categoryId: s.categoryId,
   771→    serviceId: s.serviceId,
   772→    title: s.title,
   773→    description: s.description,
   774→    addressId: customer.addresses[0].id,
   775→    scheduledAt: iso(s.when),
   776→    durationMin: service.durationMin,
   777→    status: s.status,
   778→    paymentStatus: "authorized",
   779→    price,
   780→    matchScore: s.matchScore,
   781→    createdAt: iso(createdAt),
   782→    checklist: CHECKLISTS[s.categoryId].map((label, i) => ({
   783→      id: `chk-${s.id}-${i}`,

   785→      done: ["completed", "awaiting_confirmation"].includes(s.status),
   786→    })),

   799→      const ceiling = Date.now() - 30 * 60000;
   800→      const shift = newest > ceiling ? newest - ceiling : 0;

   802→        id: `msg-${s.id}-${i}`,
   803→        authorRole: m.role,
