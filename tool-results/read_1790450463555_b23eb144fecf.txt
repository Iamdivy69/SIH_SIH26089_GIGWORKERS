    73→/* ------------------------------------------------------------------ */
    74→/* Flow state (survives navigation away and back within the session)    */

    76→
    77→interface FlowState {
    78→  categoryId: ServiceCategoryId | null;
    79→  serviceId: string | null;
    80→  description: string;
    81→  addressId: string;
    82→  notes: string;
    83→  selectedWorkerId: string | null;
    84→  slotDate: string | null;
    85→  slotTime: string | null;
    86→  paymentMethod: string;
    87→  descriptionTouched: boolean;
    88→}
    89→
    90→interface MatchCache {
    91→  key: string;
    92→  recommendations: WorkerRecommendation[];
    93→  weights: Record<string, number>;
    94→  provisionalAt: string;
    95→}
    96→
    97→const FRESH_STATE: FlowState = {
    98→  categoryId: null,
    99→  serviceId: null,
   100→  description: "",
   101→  addressId: DEMO_ADDRESSES[0].id,
   102→  notes: "",
   103→  selectedWorkerId: null,
   104→  slotDate: null,
   105→  slotTime: null,
   106→  paymentMethod: PAYMENT_METHODS[0].id,
   107→  descriptionTouched: false,
   108→};
   109→
   110→/** Module-scoped snapshot so the flow survives "View profile" detours. */
   111→let flowSnapshot: { state: FlowState; match: MatchCache | null; prefillWorkerId?: string; step: number } | null = null;
   112→
   113→const WEIGHT_LABELS: Record<string, string> = {
   114→  skill: "skills",
   115→  location: "location",
   116→  availability: "availability",
   117→  rating: "rating",
   118→  experience: "experience",
   119→};
   120→
   121→function timeLabel(time: string): string {
   122→  return format(parseISO(`2000-01-01T${time}:00`), "h:mm a");
   123→}

   126→/* Screen                                                              */

   128→
   129→export function BookingFlowScreen({ categoryIdParam }: { categoryIdParam?: string }) {
   130→  const navigate = useAppStore((s) => s.navigate);
   131→
   132→  /* ----- one-time init: prefill > snapshot > route param > fresh ----- */
   133→  const [init] = useState<{ prefill: BookingPrefill | null; restored: boolean; paramCategory: ServiceCategoryId | null }>(() => {
   134→    const prefill = peekBookingPrefill();
   135→    const paramCategory = (categoryIdParam as ServiceCategoryId) ?? null;
   136→    if (prefill) return { prefill, restored: false, paramCategory };
   137→    if (flowSnapshot && (!paramCategory || flowSnapshot.state.categoryId === paramCategory)) {
   138→      return { prefill: null, restored: true, paramCategory };
   139→    }
   140→    return { prefill: null, restored: false, paramCategory };
   141→  });
   142→
   143→  /* consume the prefill mailbox once mounted (idempotent) */
   144→  useEffect(() => {
   145→    clearBookingPrefill();
   146→  }, []);
   147→
   148→  const [state, setState] = useState<FlowState>(() => {
   149→    if (init.prefill) {
   150→      return {
   151→        ...FRESH_STATE,
   152→        categoryId: init.prefill.categoryId ?? null,
   153→        serviceId: init.prefill.serviceId ?? null,
   154→      };
   155→    }
   156→    if (init.restored && flowSnapshot) return flowSnapshot.state;
   157→    return { ...FRESH_STATE, categoryId: init.paramCategory };
   158→  });
   159→  const [match, setMatch] = useState<MatchCache | null>(() => (init.restored ? flowSnapshot?.match ?? null : null));
   160→  const [prefillWorkerId] = useState<string | undefined>(() => init.prefill?.workerId ?? (init.restored ? flowSnapshot?.prefillWorkerId : undefined));
   161→  const [step, setStep] = useState<number>(() => {
   162→    if (init.prefill) return init.prefill.serviceId ? 2 : 1;
   163→    if (init.restored && flowSnapshot) return flowSnapshot.step;
   164→    return 1;
   165→  });
   166→  const [created, setCreated] = useState<Booking | null>(null);
   167→
   168→  const patch = useCallback((p: Partial<FlowState>) => setState((s) => ({ ...s, ...p })), []);
   169→  const goto = (n: number) => {
   170→    setStep(n);
   171→    window.scrollTo({ top: 0, behavior: "smooth" });
   172→  };
   173→
   174→  /* ----- persist snapshot ----- */
   175→  useEffect(() => {
   176→    if (!created) flowSnapshot = { state, match, prefillWorkerId, step };
   177→  }, [state, match, prefillWorkerId, step, created]);
   178→
   179→  const { data: categories, isLoading: categoriesLoading } = useCategories();
   180→  const categoryList: ServiceCategory[] = categories ?? [];
   181→  const category = useMemo(() => categoryList.find((c) => c.id === state.categoryId), [categoryList, state.categoryId]);
   182→  const service = useMemo(
   183→    () => category?.services.find((s) => s.id === state.serviceId) ?? null,
   184→    [category, state.serviceId],
   185→  );
   186→  /* charge is fully derived from the selected service */
   187→  const charge = service?.basePrice ?? 0;
   188→  const price = useMemo(() => computePrice(charge), [charge]);
   189→
   190→  /* ----- matching (step 3) ----- */
   191→  const matching = useMatching();
   192→  const requestedKeyRef = useRef<string | null>(null);
   193→  const matchKey = state.categoryId && state.serviceId ? `${state.categoryId}:${state.serviceId}` : null;
   194→
   195→  useEffect(() => {
   196→    if (step !== 3 || !matchKey || match?.key === matchKey || requestedKeyRef.current === matchKey) return;
   197→    requestedKeyRef.current = matchKey;
   198→    const provisionalAt = provisionalScheduledAt();
   199→    matching.mutate(
   200→      {
   201→        categoryId: state.categoryId!,
   202→        serviceId: state.serviceId!,
   203→        description: state.description,
   204→        scheduledAt: provisionalAt,
   205→        charge,
   206→      },
   207→      {
   208→        onSuccess: (data) =>
   209→          setMatch({ key: matchKey, recommendations: data.recommendations, weights: data.weights, provisionalAt }),
   210→      },
   211→    );
   212→  }, [step, matchKey, match?.key, matching, state.categoryId, state.serviceId, state.description, charge]);
   213→
   214→  const runMatch = () => {
   215→    requestedKeyRef.current = null;
   216→    if (matchKey) {
   217→      const provisionalAt = provisionalScheduledAt();
   218→      matching.mutate(
   219→        {
   220→          categoryId: state.categoryId!,
   221→          serviceId: state.serviceId!,
   222→          description: state.description,
   223→          scheduledAt: provisionalAt,
   224→          charge,
   225→        },
   226→        {
   227→          onSuccess: (data) =>
   228→            setMatch({ key: matchKey, recommendations: data.recommendations, weights: data.weights, provisionalAt }),
   229→        },
   230→      );
   231→    }
   232→  };
   233→
   234→  /* ----- selected worker resolution (derived — a directly chosen member
   235→     is auto-selected until the customer picks someone else) ----- */
   236→  const recommendations = match?.recommendations ?? [];
   237→  const directInRecs = Boolean(prefillWorkerId && recommendations.some((r) => r.worker.id === prefillWorkerId));
   238→  const { data: directProfile } = useWorkerProfile(prefillWorkerId && !directInRecs ? prefillWorkerId : undefined);
   239→  const autoSelectedId =
   240→    prefillWorkerId && (recommendations.length > 0 || directProfile) && !state.selectedWorkerId
   241→      ? prefillWorkerId
   242→      : null;
   243→  const selectedWorkerId = state.selectedWorkerId ?? autoSelectedId;
   244→  const selectedRec = recommendations.find((r) => r.worker.id === selectedWorkerId) ?? null;
   245→  const selectedWorker: Worker | undefined =
   246→    selectedRec?.worker ?? (directProfile?.worker.id === selectedWorkerId ? directProfile.worker : undefined);
   247→  const selectedScore = selectedRec?.score;
   248→
   249→  const slotLabel =
   250→    state.slotDate && state.slotTime ? `${dayChipLabel(state.slotDate)} · ${timeLabel(state.slotTime)}` : undefined;
   251→
   252→  /* ----- booking creation ----- */
   253→  const createBooking = useCreateBooking();
   254→  const pay = () => {
   255→    if (!state.categoryId || !state.serviceId || !selectedWorkerId || !state.slotDate || !state.slotTime) return;
   256→    const input: CreateBookingInput = {
   257→      workerId: selectedWorkerId,
   258→      categoryId: state.categoryId,
   259→      serviceId: state.serviceId,
   260→      description: state.description.trim(),
   261→      addressId: state.addressId,
   262→      scheduledAt: slotToISO(state.slotDate, state.slotTime),
   263→      customerNotes: state.notes.trim() || undefined,
   264→      matchScore: selectedScore,
   265→      charge,
   266→    };
   267→    createBooking.mutate(input, {
   268→      onSuccess: (data) => {
   269→        flowSnapshot = null;
   270→        setCreated(data.booking);
   271→        goto(6);
   272→        toast.success("Payment authorized", { description: "Your request has been sent to the member." });
   273→      },
   274→    });
   275→  };
   276→
   277→  /* ----- validation ----- */
   278→  const stepValid = (n: number): boolean => {
   279→    switch (n) {
   280→      case 1:
   281→        return Boolean(state.categoryId && state.serviceId && charge > 0);
   282→      case 2:
   283→        return state.description.trim().length >= 15;
   284→      case 3:
   285→        return Boolean(selectedWorkerId);
   286→      case 4:
   287→        return Boolean(state.slotDate && state.slotTime);
   288→      default:
   289→        return true;
   290→    }
   291→  };
   292→  const canContinue = stepValid(step);
   293→
   294→  const startOver = () => {
   295→    flowSnapshot = null;
   296→    setState({ ...FRESH_STATE });
   297→    setMatch(null);
   298→    setCreated(null);
   299→    goto(1);
   300→  };
   301→
   302→  if (categoriesLoading) {
   303→    return (
   304→      <PageHeader eyebrow="New request" title="Book a service" description="Loading the cooperative service catalogue…" />
   305→    );
   306→  }
   307→
   308→  return (
   309→    <div>
   310→      <PageHeader
   311→        eyebrow="New service request"
   312→        title="Book a service"
   313→        description="Six steps — service, requirements, an explainable member match, your slot, and a transparent review before you pay."
   314→        onBack={() => navigate("customer-home")}
   315→        actions={
   316→          step < 6 ? (
   317→            <Button variant="ghost" size="sm" onClick={startOver} className="text-muted-foreground">
   318→              <RotateCcw className="h-3.5 w-3.5" strokeWidth={1.9} /> Start over
   319→            </Button>
   320→          ) : undefined
   321→        }
   322→      />
   323→
   324→      <div className="rounded-lg border bg-card px-4 py-4 sm:px-5">
   325→        <FlowStepper step={step} onJump={(n) => n < step && goto(n)} />
   326→      </div>
   327→
   328→      <div className="mt-6 grid gap-6 lg:grid-cols-[300px_minmax(0,1fr)]">
   329→        <aside className="hidden lg:block">
   330→          <div className="sticky top-[76px]">
   331→            <FlowSummaryPanel
   332→              category={category}
   333→              serviceName={service?.name}
   334→              charge={charge}
   335→              worker={selectedWorker}
   336→              matchScore={selectedScore}
   337→              slotLabel={slotLabel}
   338→              addressId={state.addressId}
   339→            />
   340→          </div>
   341→        </aside>
   342→
   343→        <div className="min-w-0">
   344→          <FlowSummaryStrip
   345→            className="mb-4 lg:hidden"
   346→            category={category}
   347→            serviceName={service?.name}
   348→            charge={charge}
   349→            worker={selectedWorker}
   350→            matchScore={selectedScore}
   351→            slotLabel={slotLabel}
   352→            addressId={state.addressId}
   353→          />
   354→
   355→          {step === 1 && (
   356→            <ServiceStep
   357→              categories={categoryList}
   358→              state={state}
   359→              patch={patch}
   360→              footer={
   361→                <StepFooter
   362→                  onNext={() => goto(2)}
   363→                  nextDisabled={!canContinue}
   364→                  nextLabel="Continue to requirements"
   365→                />
   366→              }
   367→            />
   368→          )}
   369→
   370→          {step === 2 && (
   371→            <RequirementsStep
   372→              state={state}
   373→              patch={patch}
   374→              footer={
   375→                <StepFooter
   376→                  onBack={() => goto(1)}
   377→                  onNext={() => goto(3)}
   378→                  nextDisabled={!canContinue}
   379→                  nextLabel="Find my member"
   380→                />
   381→              }
   382→            />
   383→          )}
   384→
   385→          {step === 3 && (
   386→            <MatchStep
   387→              selectedWorkerId={selectedWorkerId}
   388→              patch={patch}
   389→              match={match}
   390→              matchingPending={matching.isPending}
   391→              matchingError={matching.isError}
   392→              onRetry={runMatch}
   393→              prefillWorkerId={directInRecs ? undefined : prefillWorkerId}
   394→              onViewProfile={(worker, score, factors) => {
   395→                setWorkerMatchContext({ workerId: worker.id, score, factors });
   396→                navigate("customer-worker", { workerId: worker.id });
   397→              }}
   398→              footer={
   399→                <StepFooter
   400→                  onBack={() => goto(2)}
   401→                  onNext={() => goto(4)}
   402→                  nextDisabled={!canContinue}
   403→                  nextLabel="Choose a slot"
   404→                />
   405→              }
   406→            />
   407→          )}
   408→
   409→          {step === 4 && (
   410→            <SlotStep
   411→              state={state}
   412→              patch={patch}
   413→              worker={selectedWorker}
   414→              footer={
   415→                <StepFooter
   416→                  onBack={() => goto(3)}
   417→                  onNext={() => goto(5)}
   418→                  nextDisabled={!canContinue}
   419→                  nextLabel="Review & pay"
   420→                />
   421→              }
   422→            />
   423→          )}
   424→
   425→          {step === 5 && (
   426→            <ReviewStep
   427→              state={state}
   428→              patch={patch}
   429→              category={category}
   430→              service={service}
   431→              worker={selectedWorker}
   432→              score={selectedScore}
   433→              slotLabel={slotLabel ?? ""}
   434→              price={price}
   435→              pending={createBooking.isPending}
   436→              error={createBooking.error instanceof Error ? createBooking.error.message : undefined}
   437→              onPay={pay}
   438→              onBack={() => goto(4)}
   439→            />
   440→          )}
   441→
   442→          {step === 6 && created && <ConfirmationStep booking={created} worker={selectedWorker} />}
   443→        </div>
   444→      </div>
   445→    </div>
   446→  );
   447→}

   451→/* ------------------------------------------------------------------ */
   452→
   453→function StepFooter({
   454→  onBack,
   455→  onNext,
   456→  nextDisabled,
   457→  nextLabel,
   458→}: {
   459→  onBack?: () => void;
   460→  onNext: () => void;
   461→  nextDisabled: boolean;
   462→  nextLabel: string;
   463→}) {
   464→  return (
   465→    <div className="mt-6 flex items-center justify-between gap-3 border-t pt-5">
   466→      {onBack ? (
   467→        <Button variant="ghost" size="sm" onClick={onBack} className="text-muted-foreground">
   468→          <ArrowLeft className="h-4 w-4" strokeWidth={1.9} /> Back
   469→        </Button>
   470→      ) : (
   471→        <span aria-hidden />
   472→      )}
   473→      <Button onClick={onNext} disabled={nextDisabled}>
   474→        {nextLabel}
   475→      </Button>
   476→    </div>
   477→  );
   478→}

   483→
   484→function ServiceStep({
   485→  categories,
   486→  state,
   487→  patch,
   488→  footer,
   489→}: {
   490→  categories: ServiceCategory[];
   491→  state: FlowState;
   492→  patch: (p: Partial<FlowState>) => void;
   493→  footer: React.ReactNode;
   494→}) {
   495→  const category = categories.find((c) => c.id === state.categoryId);
   496→  if (!category) {
   497→    return (
   498→      <SectionCard title="What do you need help with?" description="Choose a service category — all are covered by verified cooperative members.">
   499→        <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 xl:grid-cols-3">
   500→          {categories.map((c) => {
   501→            const Icon = CATEGORY_ICONS[c.id];
   502→            return (
   503→              <button
   504→                key={c.id}
   505→                type="button"
   506→                onClick={() => patch({ categoryId: c.id, serviceId: null })}
   507→                className="group flex flex-col rounded-lg border bg-card p-4 text-left transition-colors hover:border-primary/40 hover:bg-muted/40 focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary"
   508→              >
   509→                <span className="flex h-9 w-9 items-center justify-center rounded-md border bg-muted/50 text-foreground">
   510→                  <Icon className="h-[18px] w-[18px]" strokeWidth={1.9} />
   511→                </span>
   512→                <span className="mt-3 text-[15px] font-semibold leading-tight">{c.name}</span>
   513→                <span className="mt-1 text-[13px] leading-snug text-muted-foreground">{c.tagline}</span>
   514→                <span className="tnum mt-3 text-xs text-muted-foreground">
   515→                  From {money(Math.min(...c.services.map((s) => s.basePrice)))} · {c.services.length} services
   516→                </span>
   517→              </button>
   518→            );
   519→          })}
   520→        </div>
   521→        {footer}
   522→      </SectionCard>
   523→    );
   524→  }
   525→
   526→  return (
   527→    <SectionCard
   528→      title={`${category.name} — choose a service`}
   529→      description={category.description}
   530→      actions={
   531→        <Button variant="ghost" size="sm" onClick={() => patch({ categoryId: null, serviceId: null })} className="text-muted-foreground">
   532→          Change category
   533→        </Button>
   534→      }
   535→    >
   536→      <fieldset className="space-y-2.5">
   537→        <legend className="sr-only">Services in {category.name}</legend>
   538→        {category.services.map((s) => {
   539→          const active = state.serviceId === s.id;
   540→          return (
   541→            <label
   542→              key={s.id}
   543→              className={cn(
   544→                "flex cursor-pointer items-start gap-3 rounded-lg border p-4 transition-colors",
   545→                active ? "border-primary bg-[oklch(0.975_0.012_155)]" : "hover:border-primary/30 hover:bg-muted/40",
   546→              )}
   547→            >
   548→              <input
   549→                type="radio"
   550→                name="service"
   551→                className="sr-only"
   552→                checked={active}
   553→                onChange={() => patch({ serviceId: s.id })}
   554→              />
   555→              <span
   556→                aria-hidden
   557→                className={cn(
   558→                  "mt-0.5 flex h-4 w-4 shrink-0 items-center justify-center rounded-full border",
   559→                  active ? "border-primary" : "border-muted-foreground/40",
   560→                )}
   561→              >
   562→                {active && <span className="h-2 w-2 rounded-full bg-primary" />}
   563→              </span>
   564→              <span className="min-w-0 flex-1">
   565→                <span className="flex flex-wrap items-baseline justify-between gap-x-4 gap-y-1">
   566→                  <span className="text-[14px] font-medium">{s.name}</span>
   567→                  <span className="tnum text-[14px] font-semibold">{money(s.basePrice)}</span>
   568→                </span>
   569→                <span className="mt-0.5 block text-[13px] leading-snug text-muted-foreground">{s.description}</span>
   570→                <span className="tnum mt-1.5 block text-xs text-muted-foreground">
   571→                  {duration(s.durationMin)} · per {s.unit}
   572→                </span>
   573→              </span>
   574→            </label>
   575→          );
   576→        })}
   577→      </fieldset>
   578→      {footer}
   579→    </SectionCard>
   580→  );
   581→}

   583→/* ------------------------------------------------------------------ */
   584→/* Step 2 — Requirements                                               */
   585→/* ------------------------------------------------------------------ */
   586→
   587→function RequirementsStep({
   588→  state,
   589→  patch,
   590→  footer,
   591→}: {
   592→  state: FlowState;
   593→  patch: (p: Partial<FlowState>) => void;
   594→  footer: React.ReactNode;
   595→}) {
   596→  const descLength = state.description.trim().length;
   597→  const descValid = descLength >= 15;
   598→  return (
   599→    <SectionCard title="Describe the work" description="This is shared with matched members — the more specific, the better the first-visit fix rate.">
   600→      <div className="space-y-6">
   601→        <div>
   602→          <Label htmlFor="bf-description" className="text-[13px] font-medium">
   603→            What is the issue? <span className="text-destructive">*</span>
   604→          </Label>
   605→          <Textarea
   606→            id="bf-description"
   607→            value={state.description}
