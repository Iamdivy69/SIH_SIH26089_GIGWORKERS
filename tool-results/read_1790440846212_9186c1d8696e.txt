    41→type Handler = (req: NextRequest, params: Params, body: any, user: string) => Promise<unknown> | unknown;
    42→
    43→const routes: { method: string; pattern: RegExp; keys: string[]; handler: Handler }[] = [];
    44→
    45→function route(method: string, path: string, handler: Handler) {
    46→  const keys: string[] = [];
    47→  const pattern = new RegExp(
    48→    "^" +
    49→      path
    50→        .split("/")
    51→        .map((seg) => {
    52→          if (seg.startsWith(":")) {
    53→            keys.push(seg.slice(1));
    54→            return "([^/]+)";
    55→          }
    56→          return seg.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    57→        })
    58→        .join("/") +
    59→      "$",
    60→  );
    61→  routes.push({ method, pattern, keys, handler });
    62→}
    63→
    64→export async function handle(req: NextRequest, slug: string[]): Promise<Response> {
    65→  await new Promise((r) => setTimeout(r, LATENCY));
    66→  const path = "/" + slug.join("/");
    67→  const method = req.method.toUpperCase();
    68→  const user = req.headers.get("x-demo-user") || "";
    69→
    70→  let body: any = null;
    71→  if (method === "POST" || method === "PUT") {
    72→    try {
    73→      body = await req.json();
    74→    } catch {
    75→      body = {};
    76→    }
    77→  }
    78→
    79→  for (const r of routes) {
    80→    if (r.method !== method) continue;
    81→    const match = path.match(r.pattern);
    82→    if (!match) continue;
    83→    const params: Params = {};
    84→    r.keys.forEach((k, i) => (params[k] = decodeURIComponent(match[i + 1])));
    85→    try {
    86→      const result = await r.handler(req, params, body, user || demoUserFor(path));
    87→      return Response.json(result as any, { status: 200 });
    88→    } catch (err: any) {
    89→      return Response.json({ error: err?.message ?? "Request failed" }, { status: 400 });
    90→    }
    91→  }
    92→  return Response.json({ error: "Not found", path }, { status: 404 });
    93→}

   185→route("GET", "/workers", (req) => {
   186→  const store = getStore();
   187→  const url = new URL(req.url);
   188→  const q = (url.searchParams.get("q") ?? "").toLowerCase();
   189→  const category = url.searchParams.get("category");
   190→  const minRating = parseFloat(url.searchParams.get("minRating") ?? "0");
   191→  const maxDistance = parseFloat(url.searchParams.get("maxDistance") ?? "999");
   192→  const verifiedOnly = url.searchParams.get("verifiedOnly") === "true";
   193→  const availableToday = url.searchParams.get("availableToday") === "true";
   194→  const sort = url.searchParams.get("sort") ?? "recommended";
   195→
   196→  let list = store.workers.filter((w) => w.status === "verified" || w.status === "suspended");
   197→  if (q) list = list.filter((w) => [w.name, w.tradeTitle, w.locality, ...w.skills].join(" ").toLowerCase().includes(q));
   198→  if (category && category !== "all") list = list.filter((w) => w.category === category);
   199→  if (minRating) list = list.filter((w) => w.rating >= minRating);
   200→  if (maxDistance < 999) list = list.filter((w) => w.distanceKm <= maxDistance);
   201→  if (verifiedOnly) list = list.filter((w) => w.status === "verified");
   202→  const today = new Date().getDay();
   203→  if (availableToday) list = list.filter((w) => w.availability.some((a) => a.day === today && a.slots.length > 0));
   204→
   205→  if (sort === "rating") list.sort((a, b) => b.rating - a.rating);
   206→  else if (sort === "distance") list.sort((a, b) => a.distanceKm - b.distanceKm);
   207→  else if (sort === "jobs") list.sort((a, b) => b.completedJobs - a.completedJobs);
   208→  else list.sort((a, b) => b.rating * 20 + b.completedJobs / 20 - (a.rating * 20 + a.completedJobs / 20));
   209→
   210→  return { items: list, total: list.length };
   211→});
